    function changeLanguage(langKey) {
      currentLang = langKey;
      const t = translations[langKey];

      // Actualizar elementos de texto con data-i18n
      document.querySelectorAll('[data-i18n]').forEach(el => {
        const key = el.getAttribute('data-i18n');
        if (t[key]) el.textContent = t[key];
      });

      // Actualizar atutributos title con data-i18n-title
      document.querySelectorAll('[data-i18n-title]').forEach(el => {
        const key = el.getAttribute('data-i18n-title');
        if (t[key]) el.title = t[key];
      });

      // Actualizar placeholders
      urlInput.placeholder = t.urlPlaceholder;
      changeSearchEngine(currentEngine);

      // Actualizar nombres de pestañas predeterminadas no renombradas
      tabs.forEach(tab => {
        if (tab.title === translations.es.newTabDefault || tab.title === translations.en.newTabDefault) {
          tab.title = t.newTabDefault;
        }
      });
      renderTabs();
    }

    function changeSearchEngine(engineKey) {
      currentEngine = engineKey;
      const engine = searchEngines[engineKey];
      const t = translations[currentLang];
      searchInput.placeholder = t.searchPlaceholder.replace('{engine}', engine.name);
    }

    function createNewTab(input = '') {
      tabIdCounter++;
      const id = tabIdCounter;

      const iframe = document.createElement('iframe');
      iframe.className = 'tab-iframe';
      iframe.id = `iframe-${id}`;
      iframesContainer.appendChild(iframe);

      const tabObj = {
        id: id,
        url: '',
        title: translations[currentLang].newTabDefault,
        iframeEl: iframe
      };

      tabs.push(tabObj);
      renderTabs();
      switchTab(id);

      if (input) {
        openUrl(input);
      }
    }

    function switchTab(id) {
      activeTabId = id;
      const tab = tabs.find(t => t.id === id);
      if (!tab) return;

      document.querySelectorAll('.tab-iframe').forEach(f => f.style.display = 'none');

      if (!tab.url) {
        homeScreen.style.display = 'flex';
        urlInput.value = '';
      } else {
        homeScreen.style.display = 'none';
        tab.iframeEl.style.display = 'block';
        urlInput.value = tab.url;
      }

      renderTabs();
    }

    function closeTab(id, event) {
      event.stopPropagation();
      
      const tabIndex = tabs.findIndex(t => t.id === id);
      if (tabIndex === -1) return;

      const tab = tabs[tabIndex];
      if (tab.iframeEl) tab.iframeEl.remove();

      tabs.splice(tabIndex, 1);

      if (tabs.length === 0) {
        createNewTab();
      } else if (activeTabId === id) {
        const nextActive = tabs[Math.max(0, tabIndex - 1)];
        switchTab(nextActive.id);
      } else {
        renderTabs();
      }
    }

    function renderTabs() {
      tabsContainer.innerHTML = '';
      tabs.forEach(tab => {
        const tabEl = document.createElement('div');
        tabEl.className = `tab ${tab.id === activeTabId ? 'active' : ''}`;
        tabEl.onclick = () => switchTab(tab.id);

        const titleEl = document.createElement('span');
        titleEl.className = 'tab-title';
        titleEl.textContent = tab.title;

        const closeBtn = document.createElement('span');
        closeBtn.className = 'close-tab';
        closeBtn.innerHTML = '✕';
        closeBtn.onclick = (e) => closeTab(tab.id, e);

        tabEl.appendChild(titleEl);
        tabEl.appendChild(closeBtn);
        tabsContainer.appendChild(tabEl);
      });
    }

    // Generador de la página de resultados propia de OSC Search

    function openUrl(input) {
      const activeTab = tabs.find(t => t.id === activeTabId);
      if (!activeTab || !input) return;

      const trimmed = input.trim();
      let targetUrl = '';

      if (/^https?:\/\//i.test(trimmed)) {
        targetUrl = trimmed;
      } else if (!trimmed.includes(' ') && trimmed.includes('.')) {
        targetUrl = 'https://' + trimmed;
      } else {
        const engine = searchEngines[currentEngine] || searchEngines.osc;
        if (currentEngine === 'osc') {
          activeTab.url = 'osc://search?q=' + encodeURIComponent(trimmed);
          activeTab.title = `⚡ ${trimmed}`;
          activeTab.iframeEl.removeAttribute('src');
          activeTab.iframeEl.srcdoc = generateOscSearchHTML(trimmed);
          
          homeScreen.style.display = 'none';
          activeTab.iframeEl.style.display = 'block';
          urlInput.value = activeTab.url;
          renderTabs();
          return;
        } else {
          targetUrl = engine.url + encodeURIComponent(trimmed);
        }
      }

      activeTab.url = targetUrl;
      activeTab.title = getDomainOrQuery(input, targetUrl);
      activeTab.iframeEl.removeAttribute('srcdoc');
      activeTab.iframeEl.src = targetUrl;

      homeScreen.style.display = 'none';
      activeTab.iframeEl.style.display = 'block';
      urlInput.value = targetUrl;

      renderTabs();
    }

    function navigate() {
      if (urlInput.value.trim()) {
        openUrl(urlInput.value.trim());
      }
    }

    function reloadTab() {
      const activeTab = tabs.find(t => t.id === activeTabId);
      if (activeTab && activeTab.url) {
        if (activeTab.iframeEl.srcdoc) {
          const content = activeTab.iframeEl.srcdoc;
          activeTab.iframeEl.srcdoc = '';
          activeTab.iframeEl.srcdoc = content;
        } else {
          activeTab.iframeEl.src = activeTab.url;
        }
      }
    }

    function goHome() {
      const activeTab = tabs.find(t => t.id === activeTabId);
      if (activeTab) {
        activeTab.url = '';
        activeTab.title = translations[currentLang].newTabDefault;
        activeTab.iframeEl.style.display = 'none';
        activeTab.iframeEl.removeAttribute('srcdoc');
        activeTab.iframeEl.src = '';
        homeScreen.style.display = 'flex';
        urlInput.value = '';
        renderTabs();
      }
    }

    function getDomainOrQuery(originalInput, fullUrl) {
      try {
        const engine = searchEngines[currentEngine];
        if (engine && engine.url !== 'internal' && fullUrl.includes(new URL(engine.url).hostname)) {
          return `${engine.icon} ${originalInput}`;
        }
        const parsed = new URL(fullUrl);
        return parsed.hostname.replace('www.', '');
      } catch {
        return originalInput;
      }
    }

    /* Ajustes Visuales de Fondo */
    function setBgFile(event) {
      const file = event.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
          homeScreen.style.backgroundImage = `url('${e.target.result}')`;
        };
        reader.readAsDataURL(file);
      }
    }

    function setBgUrl() {
      const url = prompt(translations[currentLang].promptBgUrl);
      if (url && url.trim() !== '') {
        homeScreen.style.backgroundImage = `url('${url.trim()}')`;
      }
    }

    function resetBg() {
      homeScreen.style.backgroundImage = "url('fondo.jpg')";
    }

    /* Personalización de color */
    function updateMenuColor(color) {
      document.documentElement.style.setProperty('--menu-bg', color);

    function updateAccentColor(color) {
